-- ======================== UTILITÁRIOS ========================
local function clean(s)
    return (tostring(s or ""):gsub("<[^>]+>", ""):gsub("^%s+", ""):gsub("%s+$", ""))
end

local function norm(s)
    return (clean(s):lower())
end

local KnownSet = {}
for _, k in ipairs(CONFIG.KnownRarities) do
    KnownSet[norm(k)] = true
end

local function stripLabel(text)
    text = clean(text)
    local after = text:match("^[^:]+:%s*(.+)$")
    return after and clean(after) or text
end

local function textHas(text, list)
    text = tostring(text or ""):lower()
    for _, k in ipairs(list) do
        if text:find(k, 1, true) then
            return true
        end
    end
    return false
end

local function getCharRoot()
    local char = LocalPlayer.Character
    if not char then return nil end
    local hum = char:FindFirstChildOfClass("Humanoid")
    local root = char:FindFirstChild("HumanoidRootPart")
    if hum and hum.Health > 0 and root then
        return root, hum, char
    end
    return nil
end

local function getModelRoot(model)
    return model.PrimaryPart
        or model:FindFirstChild("HumanoidRootPart")
        or model:FindFirstChildWhichIsA("BasePart", true)
end

local function getRarity(model)
    for _, key in ipairs(CONFIG.RarityKeys) do
        local attr = model:GetAttribute(key)
        if attr ~= nil and clean(attr) ~= "" then
            return clean(attr)
        end
    end

    for _, d in ipairs(model:GetDescendants()) do
        local lname = d.Name:lower()
        if lname:find("rarity", 1, true) or lname:find("raridade", 1, true) then
            if d:IsA("StringValue") and d.Value ~= "" then
                return clean(d.Value)
            elseif (d:IsA("TextLabel") or d:IsA("TextButton")) and clean(d.Text) ~= "" then
                return stripLabel(d.Text)
            end
            for _, key in ipairs(CONFIG.RarityKeys) do
                local a = d:GetAttribute(key)
                if a ~= nil and clean(a) ~= "" then
                    return clean(a)
                end
            end
        end
        if d:IsA("TextLabel") or d:IsA("TextButton") then
            local t = stripLabel(d.Text)
            if KnownSet[norm(t)] then
                return t
            end
        end
    end

    local parent = model.Parent
    if parent and parent ~= Workspace and KnownSet[norm(parent.Name)] then
        return clean(parent.Name)
    end
    return CONFIG.UnknownRarity
end

local function findPrompt(model)
    local best, bestScore = nil, 0
    for _, d in ipairs(model:GetDescendants()) do
        if d:IsA("ProximityPrompt") then
            local txt = d.ActionText .. " " .. d.ObjectText .. " " .. d.Name
            if not textHas(txt, CONFIG.BlockKeywords) then
                local score = textHas(txt, CONFIG.PromptKeywords) and 2 or 1
                if d.Enabled then
                    score += 0.5
                end
                if score > bestScore then
                    best, bestScore = d, score
                end
            end
        end
    end
    return best
end

local function getPromptPosition(prompt, model)
    local p = prompt and prompt.Parent
    if p then
        if p:IsA("BasePart") then
            return p.Position
        elseif p:IsA("Attachment") then
            return p.WorldPosition
        elseif p:IsA("Model") then
            local r = getModelRoot(p)
            if r then return r.Position end
        end
    end
    local r = getModelRoot(model)
    return r and r.Position or nil
end

local function triggerPrompt(prompt, model)
    local oldHold = prompt.HoldDuration
    local oldLos = prompt.RequiresLineOfSight
    local oldDist = prompt.MaxActivationDistance
    local startText = prompt.ActionText

    pcall(function()
        prompt.HoldDuration = 0
        prompt.RequiresLineOfSight = false
        prompt.MaxActivationDistance = math.max(oldDist, 20)
    end)

    local fired = 0
    for i = 1, CONFIG.PromptRetries do
        if typeof(fireproximityprompt) == "function" then
            pcall(fireproximityprompt, prompt)
        else
            pcall(function()
                prompt:InputHoldBegin()
                task.wait(math.max(oldHold, 0) + 0.1)
                prompt:InputHoldEnd()
            end)
        end
        fired += 1
        dbg("prompt disparado", i, prompt:GetFullName())
        task.wait(0.12)
        if not model:IsDescendantOf(Workspace)
            or not prompt:IsDescendantOf(Workspace)
            or not prompt.Enabled
            or prompt.ActionText ~= startText then
            break
        end
    end

    pcall(function()
        prompt.HoldDuration = oldHold
        prompt.RequiresLineOfSight = oldLos
        prompt.MaxActivationDistance = oldDist
    end)
    return fired > 0
end

-- ======================== TWEEN ========================
local function tweenTo(targetCF)
    local root, hum, char = getCharRoot()
    if not root then return false end

    local parts = {}
    for _, p in ipairs(char:GetDescendants()) do
        if p:IsA("BasePart") then
            parts[#parts + 1] = p
        end
    end

    local dist = (root.Position - targetCF.Position).Magnitude
    local duration = math.max(dist / CONFIG.TweenSpeed, 0.03)
    local tween = TweenService:Create(root, TweenInfo.new(duration, Enum.EasingStyle.Linear), { CFrame = targetCF })

    local finished = false
    local doneConn = tween.Completed:Connect(function()
        finished = true
    end)

    local oldPlatform = hum.PlatformStand
    hum.PlatformStand = true
    local stepConn = RunService.Stepped:Connect(function()
        for _, p in ipairs(parts) do
            if p.Parent then
                p.CanCollide = false
            end
        end
        root.AssemblyLinearVelocity = Vector3.zero
        root.AssemblyAngularVelocity = Vector3.zero
    end)

    tween:Play()
    local started = os.clock()
    while not finished and State.Alive do
        if hum.Health <= 0 or not root.Parent then break end
        if os.clock() - started > duration + 3 then break end
        RunService.Heartbeat:Wait()
    end
    if not finished then
        tween:Cancel()
    end

    doneConn:Disconnect()
    stepConn:Disconnect()
    if hum.Parent then
        hum.PlatformStand = oldPlatform
    end
    if root.Parent then
        root.AssemblyLinearVelocity = Vector3.zero
    end
    return finished
end

local function getRarityList()
    local seen, list = {}, {}

        local k = norm(e.Rarity)

    return list
end

-- ======================== ROUBO ========================
local function steal(entry)
    if State.Busy or not State.Alive then return end
    local model = entry.Model
    if not model or not model:IsDescendantOf(Workspace) then return end

    local myRoot = getCharRoot()
    if not myRoot then
        setStatus("Personagem indisponível")
        return
    end

    State.Busy = true
    Drop.Visible = false
    if not State.Base then
        State.Base = myRoot.CFrame
    end
    setStatus("Indo até " .. entry.Name)

    local stolen = false
    local ok, err = pcall(function()
        local prompt = findPrompt(model)
        if not prompt then
            error("alvo sem prompt de roubo", 0)
        end

        local reached = false
        for _ = 1, 6 do
            if not model:IsDescendantOf(Workspace) then
                error("alvo sumiu", 0)
            end
            if not prompt:IsDescendantOf(Workspace) then
                prompt = findPrompt(model)
                if not prompt then error("prompt sumiu", 0) end
            end

            local pos = getPromptPosition(prompt, model)
            if not pos then
                error("sem posição do alvo", 0)
            end
            if not tweenTo(CFrame.new(pos + Vector3.new(0, 1.5, 0))) then
                error("movimento interrompido", 0)
            end

            local r = getCharRoot()
            local np = getPromptPosition(prompt, model)
            if r and np and (r.Position - np).Magnitude <= CONFIG.ReachDistance then
                reached = true
                break
            end
        end
        if not reached then
            error("não alcançou o alvo", 0)
        end

        setStatus("Roubando " .. entry.Name)
        stolen = triggerPrompt(prompt, model)
    end)

    -- Fuga imediata: volta pra base sem esperar nada (o boss começa a perseguir ao roubar)
    setStatus("Fugindo pra base...")
    if State.Alive and State.Base then
        for _ = 1, 3 do
            if not getCharRoot() then break end
            if tweenTo(State.Base) then break end
        end
    end

    State.Cooldown[model] = os.clock() + CONFIG.FailCooldown
    if ok and stolen then
        State.Done[model] = true
    end
    State.Busy = false

    if State.Alive then
        setStatus(ok and ("Roubo enviado: " .. entry.Name) or ("Falhou: " .. tostring(err)))
    end
    dbg("steal", entry.Name, ok, err)
end

local function buildPhoto(vp, model)
    local ok, err = pcall(function()
        local prev = model.Archivable
        model.Archivable = true
        local clone = model:Clone()
        model.Archivable = prev
        if not clone then error("clone falhou") end

        for _, d in ipairs(clone:GetDescendants()) do
            if d:IsA("BaseScript") or d:IsA("ProximityPrompt") or d:IsA("BillboardGui")
                or d:IsA("Sound") or d:IsA("ParticleEmitter") then
                d:Destroy()
            end
        end
        clone.Parent = vp

        local cf, size = clone:GetBoundingBox()
        local cam = Instance.new("Camera")
        local dist = math.max(size.Magnitude, 1) * 1.1
        local camPos = cf.Position + cf.LookVector * dist + Vector3.new(0, size.Y * 0.1, 0)
        cam.CFrame = CFrame.new(camPos, cf.Position)
        cam.Parent = vp
        vp.CurrentCamera = cam
    end)
    if not ok then
        dbg("foto falhou", err)
        new("TextLabel", {
            Size = UDim2.fromScale(1, 1),
            BackgroundTransparency = 1,
            Text = "?",
            TextColor3 = COLORS.LightPurple,
            Font = Enum.Font.GothamBold,
            TextSize = 24,
        }, vp)
    end
end

local function removeEntry(model)
    local e = State.Entries[model]
    if not e then return end
    State.Entries[model] = nil
    State.Cooldown[model] = nil
    State.Done[model] = nil
    State.Count = State.Count - 1
    if e.Row then e.Row:Destroy() end
end

local function addEntry(model)
    if State.Entries[model] or not model:IsDescendantOf(Workspace) then return end

    local entry = { Model = model, Name = model.Name, Rarity = getRarity(model) }

    local row = new("Frame", {

local function hasPrefix(name)
    return name:lower():sub(1, #CONFIG.Prefix) == CONFIG.Prefix
end

local function considerInstance(inst)
    if not inst:IsA("Model") or not hasPrefix(inst.Name) then return end
    if Players:GetPlayerFromCharacter(inst) then return end

    -- ignora sub-modelos de um Exu (acessórios etc.)
    local p = inst.Parent
    while p and p ~= Workspace do
        if p:IsA("Model") and hasPrefix(p.Name) then return end
        p = p.Parent
    end

    task.delay(0.3, function()
        if not State.Alive or not inst:IsDescendantOf(Workspace) then return end
        for _, c in ipairs(inst:GetChildren()) do
            if c:IsA("Model") and hasPrefix(c.Name) then return end -- é um container
        end
        if not inst:FindFirstChildWhichIsA("BasePart", true) then return end
        addEntry(inst)
    end)
end

for _, d in ipairs(Workspace:GetDescendants()) do
    considerInstance(d)
end
connect(Workspace.DescendantAdded, considerInstance)

-- ======================== LOOPS ========================
-- Atualiza raridade, contador e valida entradas
task.spawn(function()
    while State.Alive do
        for model, e in pairs(State.Entries) do
            if not model:IsDescendantOf(Workspace) then
                removeEntry(model)
            else
                local r = getRarity(model)
                if r ~= e.Rarity then
                    e.Rarity = r
                    e.RarityLabel.Text = "Raridade: " .. r
                end
            end
        end
        CountLabel.Text = "Exus no mapa: " .. State.Count
        task.wait(1.5)
    end
end)

-- Farm AFK
task.spawn(function()
    while State.Alive do
        task.wait(0.4)
        if State.Farm and State.FarmRarity and not State.Busy then
            local myRoot = getCharRoot()
            if myRoot then
                local best, bestDist
                local now = os.clock()
                for model, e in pairs(State.Entries) do
                    if norm(e.Rarity) == norm(State.FarmRarity)
                        and not State.Done[model]
                        and (State.Cooldown[model] or 0) <= now
                        and model:IsDescendantOf(Workspace) then
                        local prompt = findPrompt(model)
                        if prompt and prompt.Enabled then
                            local pos = getPromptPosition(prompt, model)
                            if pos then
                                local d = (pos - myRoot.Position).Magnitude
                                if not bestDist or d < bestDist then
                                    best, bestDist = e, d
                                end
                            end
                        end
                    end
                end
                if best then
                    steal(best)
                else
                    setStatus("Farm AFK: aguardando " .. State.FarmRarity)
                end
            end
        end
    end
end)

local function unload()
    State.Alive = false
    State.Farm = false
    for _, c in ipairs(Connections) do
        pcall(function() c:Disconnect() end)
    end
    pcall(function() ScreenGui:Destroy() end)
    genv.AkumaZyroxUnload = nil
end
